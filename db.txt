-- ============================================================
-- AMDP Employee Panel Expansion — Chunk 1: Schema
-- Additive only. No existing tables are modified except
-- the single ALTER TABLE at the bottom.
-- ============================================================


-- ------------------------------------------------------------
-- 1. product_types
--    Defines each non-paper product category the business offers.
--    Admin can add new types here without touching code.
-- ------------------------------------------------------------
CREATE TABLE `product_types` (
  `id`          INT(11)      NOT NULL AUTO_INCREMENT,
  `name`        VARCHAR(100) NOT NULL,              -- e.g. "T-Shirt", "Mug", "Umbrella"
  `icon`        VARCHAR(100) DEFAULT NULL,          -- optional icon class or filename
  `description` TEXT         DEFAULT NULL,
  `is_active`   TINYINT(1)   NOT NULL DEFAULT 1,    -- 1 = visible, 0 = hidden
  `sort_order`  INT(11)      NOT NULL DEFAULT 0,    -- controls display order
  `created_at`  TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP(),
  `updated_at`  TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP() ON UPDATE CURRENT_TIMESTAMP(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_product_type_name` (`name`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;


-- ------------------------------------------------------------
-- 2. product_type_fields
--    Defines what fields each product type needs on the job order form.
--    field_type controls what kind of input is rendered.
-- ------------------------------------------------------------
CREATE TABLE `product_type_fields` (
  `id`            INT(11)      NOT NULL AUTO_INCREMENT,
  `product_type_id` INT(11)    NOT NULL,
  `field_name`    VARCHAR(100) NOT NULL,   -- internal key, e.g. "shirt_size"
  `field_label`   VARCHAR(100) NOT NULL,   -- displayed label, e.g. "Shirt Size"
  `field_type`    ENUM(
                    'text',
                    'number',
                    'dropdown',
                    'checkbox',
                    'textarea'
                  ) NOT NULL DEFAULT 'text',
  `is_required`   TINYINT(1)   NOT NULL DEFAULT 0,
  `sort_order`    INT(11)      NOT NULL DEFAULT 0,
  `created_at`    TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_field_per_type` (`product_type_id`, `field_name`),
  CONSTRAINT `fk_ptf_product_type`
    FOREIGN KEY (`product_type_id`)
    REFERENCES `product_types` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;


-- ------------------------------------------------------------
-- 3. product_type_field_options
--    Stores the choices for dropdown fields.
--    e.g. field "shirt_size" → options: S, M, L, XL, XXL
-- ------------------------------------------------------------
CREATE TABLE `product_type_field_options` (
  `id`        INT(11)      NOT NULL AUTO_INCREMENT,
  `field_id`  INT(11)      NOT NULL,
  `label`     VARCHAR(100) NOT NULL,   -- displayed text, e.g. "Extra Large"
  `value`     VARCHAR(100) NOT NULL,   -- stored value, e.g. "XL"
  `sort_order` INT(11)     NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  CONSTRAINT `fk_ptfo_field`
    FOREIGN KEY (`field_id`)
    REFERENCES `product_type_fields` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;


-- ------------------------------------------------------------
-- 4. product_type_pricing
--    Price per piece per product type.
--    Optionally scoped to a specific variant (e.g. XL costs more).
--    If variant_field_id and variant_value are NULL = base price.
-- ------------------------------------------------------------
CREATE TABLE `product_type_pricing` (
  `id`               INT(11)        NOT NULL AUTO_INCREMENT,
  `product_type_id`  INT(11)        NOT NULL,
  `variant_field_id` INT(11)        DEFAULT NULL,   -- which field drives the price difference
  `variant_value`    VARCHAR(100)   DEFAULT NULL,   -- which value of that field
  `price_per_piece`  DECIMAL(10,2)  NOT NULL DEFAULT 0.00,
  `effective_date`   DATE           NOT NULL DEFAULT (CURDATE()),
  `created_at`       TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP(),
  PRIMARY KEY (`id`),
  CONSTRAINT `fk_ptp_product_type`
    FOREIGN KEY (`product_type_id`)
    REFERENCES `product_types` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_ptp_variant_field`
    FOREIGN KEY (`variant_field_id`)
    REFERENCES `product_type_fields` (`id`)
    ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;


-- ------------------------------------------------------------
-- 5. job_order_field_values
--    Stores the actual field values entered per job order.
--    One row per field per job order.
-- ------------------------------------------------------------
CREATE TABLE `job_order_field_values` (
  `id`           INT(11) NOT NULL AUTO_INCREMENT,
  `job_order_id` INT(11) NOT NULL,
  `field_id`     INT(11) NOT NULL,
  `field_value`  TEXT    DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_value_per_job_field` (`job_order_id`, `field_id`),
  CONSTRAINT `fk_jofv_job_order`
    FOREIGN KEY (`job_order_id`)
    REFERENCES `job_orders` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `fk_jofv_field`
    FOREIGN KEY (`field_id`)
    REFERENCES `product_type_fields` (`id`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;


-- ------------------------------------------------------------
-- 6. ALTER existing job_orders table
--    Add product_type_id — nullable so all existing paper
--    job orders remain valid (NULL = paper, old behavior).
-- ------------------------------------------------------------
ALTER TABLE `job_orders`
  ADD COLUMN `product_type_id` INT(11) DEFAULT NULL AFTER `product_id`,
  ADD CONSTRAINT `fk_jo_product_type`
    FOREIGN KEY (`product_type_id`)
    REFERENCES `product_types` (`id`)
    ON DELETE SET NULL ON UPDATE CASCADE;
